PROCEDURE SP_CIR_NEW
(    p_TERMINAL_ID IN NUMBER DEFAULT NULL,
     p_COMPANY_ID IN NUMBER DEFAULT NULL,
     p_FROM_DATE IN VARCHAR2 DEFAULT NULL,
     p_TO_DATE IN VARCHAR2 DEFAULT NULL,
	 p_CONT_NO IN VARCHAR2 DEFAULT NULL,
	 p_BL_NO IN VARCHAR2 DEFAULT NULL,
	 p_TRIP_TYPE IN VARCHAR2 DEFAULT NULL,
	 p_CUSTOMER_ID IN NUMBER DEFAULT NULL,
	 p_SERVICE_ID IN NUMBER DEFAULT NULL,
     p_REF OUT SYS_REFCURSOR
)
AS
    BEGIN
    OPEN p_REF FOR
       SELECT DISTINCT AP.CONT_NO,FC.CONT_SIZE,FC.CONT_TYPE,CMC.COUNTRY_NAME, JOB_NO,CM.CUSTOMER_NAME,AP.PARTY_INV_NO,
   BL_NO, CR_NOTE INVOICE_NOTE, TO_CHAR(LINE_HANDOVER_DATE,'DD/MM/YYYY') LINE_HANDOVER_DATE,  TO_CHAR(SAILED,'DD/MM/YYYY') SAILED,PORT,
   TO_CHAR(TRAIN_OUT_DATE,'DD/MM/YYYY') TRAIN_OUT_DATE,
                --DECODE(ID.IGM_NO,NULL,DECODE(EB.BOOKING_NO,NULL,DB.BOOKING_REF_NO,EB.BOOKING_NO),ID.IGM_NO ||'-'|| ID.LINE_ITEM)IGM_NO,
                DECODE(FC .TRIP_TYPE,'E','Export','I','Import','D','Domestic','M','Empty Return','C','Clearance','B','Back To Town','R','Reworking','O','Overseas','N','Nomination','T','Transpor') TRIP_TYPE, 
                SM.SERVICE_NAME,
                'Credit Note' INVOICE_TYPE, NVL(III.CURRENCY,'INR') CURRENCY, NVL(III.EX_RATE,1) EX_RATE,
                 INVOICE_REF_NO,II.INVOICE_NO,CR_REF_NO,
                TO_CHAR(CR_DATE,'DD/MM/YYYY') INVOICE_DATE,
                  TO_CHAR(FC.ICD_OUT_DATE,'DD/MM/YYYY') ICD_OUT_DATE,
                TO_CHAR(FC.ICD_IN_DATE,'DD/MM/YYYY') ICD_IN_DATE,
                AP.CFS,AP.POL,AP.SB_NO,TO_CHAR(AP.SB_DATE,'DD/MM/YYYY') SB_DATE,
                LINE.CUSTOMER_CODE LINE,
                   
 --ROUND(RF_AMOUNT,2) BILL_AMOUNT,
 CRD.CR_AMOUNT BILL_AMOUNT ,
               CRD.CR_TAX TAX,
                      ROUND(CRD.CR_AMOUNT+CRD.CR_TAX,2) AMOUNT
     FROM  IMP_INVOICE II
           INNER JOIN  IMP_INVOICE_ITEMS III 
           ON  II.INVOICE_NO = III.INVOICE_NO  AND III.LINE_ITEM_ID=II.LINE_ITEM_ID
           INNER JOIN  IMP_INVOICE_TAX ITI
           ON  ITI.ITEM_KEY_ID =III.ITEM_KEY_ID
               INNER JOIN CR_NOTE CR
           ON CR.INVOICE_ID=II.INVOICE_NO
            INNER JOIN CR_ITEM_DETAILS CRD
           ON CR.CR_ID=CRD.CR_ID
              INNER JOIN SERVICE_MASTER SM
           ON SM.SERVICE_ID=CRD.SERVICE_ID
               INNER JOIN CUSTOMER_MASTER CM
           ON CM.CUSTOMER_ID=II.BILL_TO
             INNER JOIN ALL_PARTY_ACCOUNT AP
           ON AP.CONT_JO_ID=III.LINE_ITEM_ID AND AP.MTY_CONT_ID=III.IMP_CONT_ID
            INNER JOIN FLEET_CONT_JO_DTLS FC
           ON AP.MTY_CONT_ID=FC.MTY_CONT_ID  
            LEFT JOIN CUSTOMER_MASTER LINE
           ON LINE.CUSTOMER_ID=FC.LINE_ID 
              LEFT JOIN PORT_MASTER PM
           ON PM.PORT_ID =AP.POD_ID   
            LEFT JOIN COUNTRY_MASTER CMC
           ON CMC.COUNTRY_ID=PM.COUNTRY_ID           
  WHERE   FC.TRIP_TYPE    = DECODE(p_TRIP_TYPE,'',FC.TRIP_TYPE,p_TRIP_TYPE)
    AND    AP.CONT_NO        = DECODE(p_CONT_NO,'',AP.CONT_NO,p_CONT_NO)
        --AND    AP.BL_NO        = DECODE(p_BL_NO,'',AP.BL_NO,p_BL_NO)
        AND    NVL (AP.BL_NO ,0)       = DECODE(p_BL_NO,'',NVL(AP.BL_NO,0),p_BL_NO)
    AND    CM.CUSTOMER_ID =DECODE(p_CUSTOMER_ID,0,CM.CUSTOMER_ID,p_CUSTOMER_ID)
    AND    CRD.SERVICE_ID  = DECODE(p_SERVICE_ID,0,SM.SERVICE_ID,p_SERVICE_ID)
  ---  AND      III.BILL_AMOUNT >0
    AND II.CANCLE_FLAGE IS NULL
    AND CANCEL_FLAG is null
    --AND III.SERVICE_ID=6
    AND CR.COMPANY_ID =p_COMPANY_ID 
    AND    CR_DATE >= DECODE(p_FROM_DATE,'',CR_DATE,TO_DATE(p_FROM_DATE,'DD/MM/YYYY'))
    AND    CR_DATE <= DECODE(p_TO_DATE,'',CR_DATE,TO_DATE(p_TO_DATE,'DD/MM/YYYY'))
   
 GROUP BY AP.CONT_NO,FC.CONT_SIZE,FC.CONT_TYPE,CMC.COUNTRY_NAME,CRD.CR_AMOUNT,CRD.CR_TAX,CR_NOTE,
  JOB_NO,CM.CUSTOMER_NAME,AP.BL_NO,SB_NO,SB_DATE,LINE_HANDOVER_DATE,
  SAILED,PORT,TRAIN_OUT_DATE, AP.PARTY_INV_NO, SERVICE_NAME,FC.TRIP_TYPE,INVOICE_NOTE, INVOICE_REF_NO,CR_REF_NO,II.INVOICE_NO,
  CR_DATE,BILL_AMOUNT,--ID.IGM_NO,ID.LINE_ITEM,EB.BOOKING_NO,DB.BOOKING_REF_NO, 
    III.BILL_QNTY,BILL_RATE,FC.ICD_OUT_DATE,FC.ICD_IN_DATE,AP.POL, LINE.CUSTOMER_CODE,AP.CFS,III.CURRENCY,III.EX_RATE
    UNION
          SELECT DISTINCT AP.CONT_NO,FC.CONT_SIZE,FC.CONT_TYPE,CMC.COUNTRY_NAME, JOB_NO,CM.CUSTOMER_NAME,AP.PARTY_INV_NO,BL_NO,II.INVOICE_NOTE, TO_CHAR(LINE_HANDOVER_DATE,'DD/MM/YYYY') LINE_HANDOVER_DATE,  TO_CHAR(SAILED,'DD/MM/YYYY') SAILED,PORT,
               TO_CHAR(TRAIN_OUT_DATE,'DD/MM/YYYY') TRAIN_OUT_DATE, --DECODE(ID.IGM_NO,NULL,DECODE(EB.BOOKING_NO,NULL,DB.BOOKING_REF_NO,EB.BOOKING_NO),ID.IGM_NO ||'-'|| ID.LINE_ITEM)IGM_NO,
                DECODE(FC .TRIP_TYPE,'E','Export','I','Import','D','Domestic','M','Empty Return','C','Clearance','B','Back To Town','R','Reworking','O','Overseas','N','Nomination','T','Transpor') TRIP_TYPE, 
                SERVICE_NAME,
                'Invoice' INVOICE_TYPE, NVL(III.CURRENCY,'INR') CURRENCY, NVL(III.EX_RATE,1) EX_RATE,
                INVOICE_REF_NO,II.INVOICE_NO,'' CR_REF_NO,
                TO_CHAR(INVOICE_DATE,'DD/MM/YYYY') INVOICE_DATE,
                  TO_CHAR(FC.ICD_OUT_DATE,'DD/MM/YYYY') ICD_OUT_DATE,
                TO_CHAR(FC.ICD_IN_DATE,'DD/MM/YYYY') ICD_IN_DATE,
                AP.CFS,AP.POL,AP.SB_NO,TO_CHAR(AP.SB_DATE,'DD/MM/YYYY') SB_DATE,
                LINE.CUSTOMER_CODE LINE,
                ROUND(BILL_QNTY*BILL_RATE,2) BILL_AMOUNT,
                ROUND(SUM(ITI.TAX_AMT) ,2) TAX,
                   ROUND( BILL_AMOUNT ,2) AS AMOUNT
     FROM  IMP_INVOICE II
           INNER JOIN  IMP_INVOICE_ITEMS III 
           ON  II.INVOICE_NO = III.INVOICE_NO  AND III.LINE_ITEM_ID=II.LINE_ITEM_ID
           INNER JOIN  IMP_INVOICE_TAX ITI
           ON  ITI.ITEM_KEY_ID =III.ITEM_KEY_ID
              INNER JOIN SERVICE_MASTER SM
           ON SM.SERVICE_ID=III.SERVICE_ID
               INNER JOIN CUSTOMER_MASTER CM
           ON CM.CUSTOMER_ID=II.BILL_TO
             INNER JOIN ALL_PARTY_ACCOUNT AP
           ON AP.CONT_JO_ID=III.LINE_ITEM_ID AND AP.MTY_CONT_ID=III.IMP_CONT_ID
            INNER JOIN FLEET_CONT_JO_DTLS FC
           ON AP.MTY_CONT_ID=FC.MTY_CONT_ID  
            LEFT JOIN CUSTOMER_MASTER LINE
           ON LINE.CUSTOMER_ID=FC.LINE_ID 
              LEFT JOIN PORT_MASTER PM
           ON PM.PORT_ID =AP.POD_ID   
            LEFT JOIN COUNTRY_MASTER CMC
           ON CMC.COUNTRY_ID=PM.COUNTRY_ID           
    WHERE   FC.TRIP_TYPE    = DECODE(p_TRIP_TYPE,'',FC.TRIP_TYPE,p_TRIP_TYPE)
    AND    AP.CONT_NO        = DECODE(p_CONT_NO,'',AP.CONT_NO,p_CONT_NO)
        --AND    AP.BL_NO        = DECODE(p_BL_NO,'',AP.BL_NO,p_BL_NO)
        AND    NVL (AP.BL_NO ,0)       = DECODE(p_BL_NO,'',NVL(AP.BL_NO,0),p_BL_NO)
    AND    CM.CUSTOMER_ID =DECODE(p_CUSTOMER_ID,0,CM.CUSTOMER_ID,p_CUSTOMER_ID)
    AND    III.SERVICE_ID  = DECODE(p_SERVICE_ID,0,SM.SERVICE_ID,p_SERVICE_ID)
    AND      III.BILL_AMOUNT >0
    AND II.CANCLE_FLAGE IS NULL
    
    --AND III.SERVICE_ID=6
    AND II.COMPANY_ID =p_COMPANY_ID 
--AND II.TERMINAL_ID=p_TERMINAL_ID
    AND    II.INVOICE_DATE >= DECODE(p_FROM_DATE,'',INVOICE_DATE,TO_DATE(p_FROM_DATE,'DD/MM/YYYY'))
    AND    II.INVOICE_DATE <= DECODE(p_TO_DATE,'',INVOICE_DATE,TO_DATE(p_TO_DATE,'DD/MM/YYYY'))
   
 GROUP BY AP.CONT_NO,FC.CONT_SIZE,FC.CONT_TYPE,CMC.COUNTRY_NAME, JOB_NO,CM.CUSTOMER_NAME,AP.BL_NO,SB_NO,SB_DATE,LINE_HANDOVER_DATE,SAILED,PORT,TRAIN_OUT_DATE, AP.PARTY_INV_NO, SERVICE_NAME,FC.TRIP_TYPE,INVOICE_NOTE, INVOICE_REF_NO,II.INVOICE_NO,INVOICE_DATE,BILL_AMOUNT,--ID.IGM_NO,ID.LINE_ITEM,EB.BOOKING_NO,DB.BOOKING_REF_NO, 
    BILL_QNTY,BILL_RATE,FC.ICD_OUT_DATE,FC.ICD_IN_DATE,AP.POL, LINE.CUSTOMER_CODE,AP.CFS,III.CURRENCY,III.EX_RATE
    
    UNION
              SELECT DISTINCT AP.CONT_NO,FC.CONT_SIZE,FC.CONT_TYPE,CMC.COUNTRY_NAME, JOB_NO,CM.CUSTOMER_NAME,AP.PARTY_INV_NO,BL_NO,II.INVOICE_NOTE, TO_CHAR(LINE_HANDOVER_DATE,'DD/MM/YYYY') LINE_HANDOVER_DATE,  TO_CHAR(SAILED,'DD/MM/YYYY') SAILED,PORT,
          TO_CHAR(TRAIN_OUT_DATE,'DD/MM/YYYY') TRAIN_OUT_DATE,      --DECODE(ID.IGM_NO,NULL,DECODE(EB.BOOKING_NO,NULL,DB.BOOKING_REF_NO,EB.BOOKING_NO),ID.IGM_NO ||'-'|| ID.LINE_ITEM)IGM_NO,
               DECODE(FC .TRIP_TYPE,'E','Export','I','Import','D','Domestic','M','Empty Return','C','Clearance','B','Back To Town','R','Reworking','O','Overseas','N','Nomination','T','Transpor') TRIP_TYPE, 
                       SERVICE_NAME,
                'Invoice' INVOICE_TYPE, NVL(III.CURRENCY,'INR') CURRENCY, NVL(III.EX_RATE,1) EX_RATE,
                INVOICE_REF_NO,II.INVOICE_NO,''CR_REF_NO,
                TO_CHAR(INVOICE_DATE,'DD/MM/YYYY') INVOICE_DATE,
                  TO_CHAR(FC.ICD_OUT_DATE,'DD/MM/YYYY') ICD_OUT_DATE,
                TO_CHAR(FC.ICD_IN_DATE,'DD/MM/YYYY') ICD_IN_DATE,
                AP.CFS,AP.POL,AP.SB_NO,TO_CHAR(AP.SB_DATE,'DD/MM/YYYY') SB_DATE,
                LINE.CUSTOMER_CODE LINE,
                ROUND(BILL_QNTY*BILL_RATE,2) BILL_AMOUNT,
                ROUND(SUM(ITI.TAX_AMT) ,2) TAX,
                   ROUND( BILL_AMOUNT ,2) AS AMOUNT
     FROM  IMP_INVOICE II
           INNER JOIN  IMP_INVOICE_ITEMS III 
           ON  II.INVOICE_NO = III.INVOICE_NO 
           INNER JOIN  IMP_INVOICE_TAX ITI
           ON  ITI.ITEM_KEY_ID =III.ITEM_KEY_ID
              INNER JOIN SERVICE_MASTER SM
           ON SM.SERVICE_ID=III.SERVICE_ID
               INNER JOIN CUSTOMER_MASTER CM
           ON CM.CUSTOMER_ID=II.BILL_TO
            INNER JOIN ALL_PARTY_ACCOUNT AP
           ON AP.CONT_JO_ID=III.LINE_ITEM_ID AND III.IMP_CONT_ID=0
            INNER JOIN FLEET_CONT_JO_DTLS FC
           ON AP.MTY_CONT_ID=FC.MTY_CONT_ID 
            LEFT JOIN CUSTOMER_MASTER LINE
           ON LINE.CUSTOMER_ID=FC.LINE_ID
              LEFT JOIN PORT_MASTER PM
           ON PM.PORT_ID =AP.POD_ID   
            LEFT JOIN COUNTRY_MASTER CMC
           ON CMC.COUNTRY_ID=PM.COUNTRY_ID             
    WHERE   FC.TRIP_TYPE    = DECODE(p_TRIP_TYPE,'',FC.TRIP_TYPE,p_TRIP_TYPE)
    AND    AP.CONT_NO        = DECODE(p_CONT_NO,'',AP.CONT_NO,p_CONT_NO)
        --AND    AP.BL_NO        = DECODE(p_BL_NO,'',AP.BL_NO,p_BL_NO)
        AND    NVL (AP.BL_NO ,0)       = DECODE(p_BL_NO,'',NVL(AP.BL_NO,0),p_BL_NO)
    AND    CM.CUSTOMER_ID =DECODE(p_CUSTOMER_ID,0,CM.CUSTOMER_ID,p_CUSTOMER_ID)
    AND    III.SERVICE_ID  = DECODE(p_SERVICE_ID,0,SM.SERVICE_ID,p_SERVICE_ID)
    AND      III.BILL_AMOUNT >0
    AND II.CANCLE_FLAGE IS NULL
    
    --AND III.SERVICE_ID=6
    AND II.COMPANY_ID =p_COMPANY_ID 
--AND II.TERMINAL_ID=p_TERMINAL_ID
    AND    II.INVOICE_DATE >= DECODE(p_FROM_DATE,'',INVOICE_DATE,TO_DATE(p_FROM_DATE,'DD/MM/YYYY'))
    AND    II.INVOICE_DATE <= DECODE(p_TO_DATE,'',INVOICE_DATE,TO_DATE(p_TO_DATE,'DD/MM/YYYY'))
 GROUP BY AP.CONT_NO,FC.CONT_SIZE,FC.CONT_TYPE,CMC.COUNTRY_NAME, JOB_NO,CM.CUSTOMER_NAME,AP.BL_NO,SB_NO,SB_DATE,LINE_HANDOVER_DATE,SAILED,PORT, TRAIN_OUT_DATE,AP.PARTY_INV_NO, SERVICE_NAME,FC.TRIP_TYPE,INVOICE_NOTE, INVOICE_REF_NO,II.INVOICE_NO,INVOICE_DATE,BILL_AMOUNT,--ID.IGM_NO,ID.LINE_ITEM,EB.BOOKING_NO,DB.BOOKING_REF_NO, 
    BILL_QNTY,BILL_RATE,FC.ICD_OUT_DATE,FC.ICD_IN_DATE,AP.POL, LINE.CUSTOMER_CODE,AP.CFS,III.CURRENCY,III.EX_RATE
    
    UNION
                                  SELECT DISTINCT '' CONT_NO,'' CONT_SIZE,'' CONT_TYPE,'' COUNTRY_NAME,
             II.BOOKING_NO JOB_NO,CM.CUSTOMER_NAME,'' PARTY_INV_NO,'' BL_NO,II.INVOICE_NOTE,
               '' LINE_HANDOVER_DATE,  '' SAILED,'' PORT,
       ''  TRAIN_OUT_DATE,      --DECODE(ID.IGM_NO,NULL,DECODE(EB.BOOKING_NO,NULL,DB.BOOKING_REF_NO,EB.BOOKING_NO),ID.IGM_NO ||'-'|| ID.LINE_ITEM)IGM_NO,
              
        'REBATE' TRIP_TYPE,
                       SERVICE_NAME,
                'Invoice' INVOICE_TYPE, NVL(III.CURRENCY,'INR') CURRENCY, NVL(III.EX_RATE,1) EX_RATE,
                INVOICE_REF_NO,II.INVOICE_NO,''CR_REF_NO,
                TO_CHAR(INVOICE_DATE,'DD/MM/YYYY') INVOICE_DATE,
             ''ICD_OUT_DATE,
               '' ICD_IN_DATE,
                '' CFS,'' POL,'' SB_NO,''  SB_DATE,
               ''  LINE,
                ROUND(BILL_QNTY*BILL_RATE,2) BILL_AMOUNT,
                ROUND(SUM(ITI.TAX_AMT) ,2) TAX,
                   ROUND( BILL_AMOUNT ,2) AS AMOUNT
     FROM  IMP_INVOICE II
           INNER JOIN  IMP_INVOICE_ITEMS III 
           ON  II.INVOICE_NO = III.INVOICE_NO 
           INNER JOIN  IMP_INVOICE_TAX ITI
           ON  ITI.ITEM_KEY_ID =III.ITEM_KEY_ID
              INNER JOIN SERVICE_MASTER SM
           ON SM.SERVICE_ID=III.SERVICE_ID
               INNER JOIN CUSTOMER_MASTER CM
           ON CM.CUSTOMER_ID=II.BILL_TO
             LEFT JOIN ALL_PARTY_ACCOUNT AP
           ON AP.CONT_JO_ID=III.LINE_ITEM_ID
            WHERE   III.BILL_AMOUNT >0
    AND II.CANCLE_FLAGE IS NULL
    AND II.SERVICE_TYPE='R'
   --AND    AP.CONT_NO        = DECODE(p_CONT_NO,'',AP.CONT_NO,p_CONT_NO)
    AND    NVL (AP.CONT_NO ,0)       = DECODE(p_CONT_NO,'',NVL(AP.CONT_NO,0),p_CONT_NO)
         AND    NVL (AP.BL_NO ,0)       = DECODE(p_BL_NO,'',NVL(AP.BL_NO,0),p_BL_NO)
 AND    CM.CUSTOMER_ID =DECODE(p_CUSTOMER_ID,0,CM.CUSTOMER_ID,p_CUSTOMER_ID)
    AND    III.SERVICE_ID  = DECODE(p_SERVICE_ID,0,SM.SERVICE_ID,p_SERVICE_ID)
   AND II.COMPANY_ID =p_COMPANY_ID 
    AND    II.INVOICE_DATE >= DECODE(p_FROM_DATE,'',INVOICE_DATE,TO_DATE(p_FROM_DATE,'DD/MM/YYYY'))
    AND    II.INVOICE_DATE <= DECODE(p_TO_DATE,'',INVOICE_DATE,TO_DATE(p_TO_DATE,'DD/MM/YYYY'))
 GROUP BY  CM.CUSTOMER_NAME, SERVICE_NAME,INVOICE_NOTE, 
 INVOICE_REF_NO,II.INVOICE_NO,INVOICE_DATE,BILL_AMOUNT,--ID.IGM_NO,ID.LINE_ITEM,EB.BOOKING_NO,DB.BOOKING_REF_NO, 
    BILL_QNTY,BILL_RATE,II.BOOKING_NO,III.CURRENCY,III.EX_RATE
    UNION
       SELECT DISTINCT '' CONT_NO,'' CONT_SIZE,'' CONT_TYPE,'' COUNTRY_NAME,
             II.BOOKING_NO JOB_NO,CM.CUSTOMER_NAME,'' PARTY_INV_NO,'' BL_NO,cr_NOTE,
               '' LINE_HANDOVER_DATE,  '' SAILED,'' PORT,
       ''  TRAIN_OUT_DATE,    
        'REBATE' TRIP_TYPE,
                       SERVICE_NAME,
                'Credit Note' INVOICE_TYPE, NVL(III.CURRENCY,'INR') CURRENCY, NVL(III.EX_RATE,1) EX_RATE,
                INVOICE_REF_NO,II.INVOICE_NO,CR_REF_NO,
                TO_CHAR(CR_DATE,'DD/MM/YYYY') INVOICE_DATE,
             ''ICD_OUT_DATE,
               '' ICD_IN_DATE,
                '' CFS,'' POL,'' SB_NO,''  SB_DATE,
               ''  LINE,
 CRD.CR_AMOUNT BILL_AMOUNT ,
               CRD.CR_TAX TAX,
                      ROUND(CRD.CR_AMOUNT+CRD.CR_TAX,2) AMOUNT
     FROM  IMP_INVOICE II
           INNER JOIN  IMP_INVOICE_ITEMS III 
           ON  II.INVOICE_NO = III.INVOICE_NO  AND III.LINE_ITEM_ID=II.LINE_ITEM_ID
 AND II.SERVICE_TYPE='R'
           INNER JOIN  IMP_INVOICE_TAX ITI
           ON  ITI.ITEM_KEY_ID =III.ITEM_KEY_ID
               INNER JOIN CR_NOTE CR
           ON CR.INVOICE_ID=II.INVOICE_NO
            INNER JOIN CR_ITEM_DETAILS CRD
           ON CR.CR_ID=CRD.CR_ID
              INNER JOIN SERVICE_MASTER SM
           ON SM.SERVICE_ID=CRD.SERVICE_ID
               INNER JOIN CUSTOMER_MASTER CM
           ON CM.CUSTOMER_ID=II.BILL_TO
                     
  WHERE 
     II.CANCLE_FLAGE IS NULL
    AND CR.COMPANY_ID =p_COMPANY_ID 
    --AND    CR_DATE >= DECODE(p_FROM_DATE,'',CR_DATE,TO_DATE(p_FROM_DATE,'DD/MM/YYYY'))
    --AND    CR_DATE <= DECODE(p_TO_DATE,'',CR_DATE,TO_DATE(p_TO_DATE,'DD/MM/YYYY'))
    AND CR_DATE >= TO_DATE(p_FROM_DATE,'DD/MM/YYYY')
 AND CR_DATE <= TO_DATE(p_TO_DATE,'DD/MM/YYYY')+1
  AND    CM.CUSTOMER_ID =DECODE(p_CUSTOMER_ID,0,CM.CUSTOMER_ID,p_CUSTOMER_ID)
    AND    CRD.SERVICE_ID  = DECODE(p_SERVICE_ID,0,SM.SERVICE_ID,p_SERVICE_ID)
 --   AND II.SERVICE_TYPE='R'
    
    AND CANCEL_FLAG is null
   
 GROUP BY CRD.CR_AMOUNT,CRD.CR_TAX,CR_NOTE,
  BOOKING_NO ,CM.CUSTOMER_NAME, SERVICE_NAME,INVOICE_NOTE, INVOICE_REF_NO,CR_REF_NO,II.INVOICE_NO,
  CR_DATE,BILL_AMOUNT,--ID.IGM_NO,ID.LINE_ITEM,EB.BOOKING_NO,DB.BOOKING_REF_NO, 
    III.BILL_QNTY,BILL_RATE,III.CURRENCY,III.EX_RATE
    
    ORDER BY INVOICE_REF_NO ASC;
  
END SP_CIR_NEW;
